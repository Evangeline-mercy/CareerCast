def skill_similarity(candidate_skill, career_skill):

    if not candidate_skill or not career_skill:
        return 0.0

    candidate_skill = normalize_skill(candidate_skill)
    career_skill = normalize_skill(career_skill)

    if not candidate_skill or not career_skill:
        return 0.0

    # ========================================================
    # 1. EXACT MATCH
    # ========================================================

    if candidate_skill == career_skill:
        return 1.0

    # ========================================================
    # 2. EXPLICIT DATASET EQUIVALENCE
    # ========================================================

    equivalents = DATASET_SKILL_EQUIVALENTS.get(
        candidate_skill,
        set()
    )

    if career_skill in equivalents:

        if candidate_skill == "sql":
            return 1.0

        if candidate_skill == "python":
            return 1.0

        if candidate_skill == "machine learning":
            return 0.60

        if candidate_skill == "data analysis":
            return 0.60

        # Libraries are supporting skills.
        # Do not treat them as direct Python matches.
        if candidate_skill in {
            "pandas",
            "numpy",
            "tensorflow",
        }:
            return 0.0

    # ========================================================
    # 3. DATA-SCIENCE / ML SEMANTIC GROUPS
    # ========================================================

    ml_terms = {
        "machine learning",
        "deep learning",
        "predictive modeling",
        "machine learning models",
        "artificial intelligence",
        "ai",
    }

    ml_dataset_terms = {
        "computer modeling software",
        "statistical analysis software",
        "data analysis software",
        "data mining software",
        "predictive analytics software",
        "statistical software",
        "modeling software",
        "data visualization software",
    }

    if candidate_skill in ml_terms:
        if career_skill in ml_dataset_terms:
            return 0.55

    # ========================================================
    # 4. DATA ANALYSIS GROUP
    # ========================================================

    data_analysis_terms = {
        "data analysis",
        "data analytics",
        "data preprocessing",
        "data visualization",
        "statistical analysis",
    }

    data_dataset_terms = {
        "statistical analysis software",
        "data analysis software",
        "data visualization software",
        "statistical software",
        "data mining software",
        "business intelligence software",
    }

    if candidate_skill in data_analysis_terms:
        if career_skill in data_dataset_terms:
            return 0.60

    # ========================================================
    # 5. PYTHON ECOSYSTEM
    # ========================================================

    python_related = {
        "python",
        "pandas",
        "numpy",
        "tensorflow",
    }

    python_dataset_terms = {
        "python",
        "oracle java",
        "software development tools",
        "programming software",
        "computer programming software",
    }

    if candidate_skill in python_related:

        if career_skill == "python":
            return 1.0

        # Pandas / NumPy / TensorFlow should NOT become
        # strong matches to generic programming skills.
        if candidate_skill in {
            "pandas",
            "numpy",
            "tensorflow",
        }:
            return 0.0

    # ========================================================
    # 6. SQL FAMILY
    # ========================================================

    sql_terms = {
        "sql",
        "structured query language",
        "structured query language sql",
    }

    if candidate_skill in sql_terms:

        if career_skill in {
            "sql",
            "structured query language",
            "structured query language sql",
            "database software",
            "database management software",
            "oracle database",
        }:
            return 0.85

    # ========================================================
    # 7. TOKEN OVERLAP
    # ========================================================

    candidate_tokens = set(
        candidate_skill.split()
    )

    career_tokens = set(
        career_skill.split()
    )

    if not candidate_tokens or not career_tokens:
        return 0.0

    intersection = (
        candidate_tokens
        &
        career_tokens
    )

    if not intersection:
        return 0.0

    candidate_coverage = (
        len(intersection)
        /
        len(candidate_tokens)
    )

    career_coverage = (
        len(intersection)
        /
        len(career_tokens)
    )

    # Require meaningful overlap.
    if candidate_coverage < 0.80:
        return 0.0

    score = (
        0.70 * candidate_coverage
        +
        0.30 * career_coverage
    )

    return min(
        1.0,
        score
    )

def calculate_skill_match(
    candidate_skills,
    career_skills,
    idf
):

    if (
        not candidate_skills
        or not career_skills
    ):

        return 0.0, []

    matched_details = []

    # --------------------------------------------------------
    # Find best career match for every candidate skill
    # --------------------------------------------------------

    for candidate_skill in (
        candidate_skills
    ):

        best_score = 0.0

        best_skill = None

        for career_skill in (
            career_skills
        ):

            similarity = (
                skill_similarity(
                    candidate_skill,
                    career_skill
                )
            )

            if similarity > best_score:

                best_score = similarity

                best_skill = career_skill

        if (
            best_skill is not None
            and best_score >= 0.80
        ):

            matched_details.append(
                (
                    candidate_skill,
                    best_skill,
                    best_score
                )
            )

    if not matched_details:

        return 0.0, []

    # --------------------------------------------------------
    # Candidate skill weights
    # --------------------------------------------------------

    candidate_weights = {}

    for skill in candidate_skills:

        candidate_weights[
            skill
        ] = idf.get(
            skill,
            1.0
        )

    total_weight = sum(
        candidate_weights.values()
    )

    if total_weight <= 0:

        return 0.0, []

    # --------------------------------------------------------
    # Weighted coverage
    # --------------------------------------------------------

    matched_weight = 0.0

    for (
        candidate_skill,
        career_skill,
        match_strength
    ) in matched_details:

        matched_weight += (

            candidate_weights[
                candidate_skill
            ]

            *

            match_strength
        )

    weighted_coverage = (

        matched_weight
        /
        total_weight
    )

    # --------------------------------------------------------
    # Simple coverage
    # --------------------------------------------------------

    simple_coverage = (

        len(matched_details)
        /
        len(candidate_skills)
    )

    # --------------------------------------------------------
    # Final score

def robust_normalize(values):

    values = np.asarray(
        values,
        dtype=np.float64
    )

    if len(values) == 0:

        return values

    low = np.percentile(
        values,
        5
    )

    high = np.percentile(
        values,
        95
    )

    if high - low < 1e-12:

        return np.zeros_like(
            values
        )

    clipped = np.clip(
        values,
        low,
        high
    )

    normalized = (

        clipped - low
    ) / (

        high - low
    )

    return normalized

        )

    else:

        confidence = (
            (scores - minimum)
            /
            (maximum - minimum)
        )

    confidence = (
        confidence * 100.0

    return confidence
